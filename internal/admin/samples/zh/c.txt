/* 构建：cc notify.c -o notify -lcurl
 *
 * C 标准库没有 HTTP 客户端，所以这里用 libcurl——多数平台本来就带着它。
 * Debian/Ubuntu 上是 libcurl4-openssl-dev；Windows 上用 vcpkg 或
 * curl-for-win 提供的 libcurl。
 *
 * libcurl 默认校验证书链。不要加 CURLOPT_SSL_VERIFYPEER(0L) 或
 * CURLOPT_SSL_VERIFYHOST(0L)：这是客户端把 bearer token 泄漏出去最常见的
 * 一种方式，而这里的 token 就走请求头。
 */
#include <curl/curl.h>
#include <stdio.h>
#include <stdlib.h>

#define BASE_URL "{{BASE_URL}}"

/* 统计响应体长度后丢弃。这里不需要 GET 的响应体；
 * 真实客户端会把它追加到缓冲区里。 */
static size_t discard(char *ptr, size_t size, size_t nmemb, void *userdata) {
    (void)ptr;
    (void)userdata;
    return size * nmemb;
}

/* 发一个请求，返回 HTTP 状态码；压根没连上时返回 -1。
 *
 * 这个区分很重要：-1 是 CONNECT_ERROR——对端根本没连上，所以这次投递没有
 * 计入渠道的配额。返回了 HTTP 状态码则说明端点答复了，不管它说了什么。 */
static long call(CURL *curl, const char *method, const char *path,
                 const char *auth, const char *body) {
    char url[1024];
    snprintf(url, sizeof url, "%s%s", BASE_URL, path);

    struct curl_slist *headers = NULL;
    headers = curl_slist_append(headers, auth);
    if (body != NULL) {
        headers = curl_slist_append(headers, "Content-Type: application/json");
        /* 用同一个 key 重发会重放第一次的响应，
         * 渠道那边仍然只看到一条消息。 */
        headers = curl_slist_append(headers, "Idempotency-Key: db-03-lag-20260922T091400");
    }

    curl_easy_setopt(curl, CURLOPT_URL, url);
    curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);
    curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, method);
    curl_easy_setopt(curl, CURLOPT_POSTFIELDS, body); /* NULL for a GET */
    curl_easy_setopt(curl, CURLOPT_TIMEOUT, 10L);
    curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, discard);
    curl_easy_setopt(curl, CURLOPT_WRITEDATA, NULL);

    long status = -1;
    CURLcode rc = curl_easy_perform(curl);
    if (rc == CURLE_OK) {
        curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, &status);
    } else {
        fprintf(stderr, "request failed: %s\n", curl_easy_strerror(rc));
    }

    curl_slist_free_all(headers);
    return status;
}

int main(void) {
    const char *token = getenv("NOTIFYRELAY_TOKEN");
    if (token == NULL || token[0] == '\0') {
        fprintf(stderr, "set NOTIFYRELAY_TOKEN first\n");
        return 1;
    }

    char auth[512];
    snprintf(auth, sizeof auth, "Authorization: Bearer %s", token);

    const char *payload =
        "{"
        "\"targets\":[\"oncall\"],"
        "\"title\":\"Replication lag on db-03\","
        "\"body\":\"**lag**: 12s\","
        "\"format\":\"markdown\","
        "\"type\":\"warning\""
        "}";

    curl_global_init(CURL_GLOBAL_DEFAULT);
    CURL *curl = curl_easy_init();
    if (curl == NULL) {
        fprintf(stderr, "curl_easy_init failed\n");
        curl_global_cleanup();
        return 1;
    }

    long status = call(curl, "POST", "/api/v1/notify", auth, payload);
    printf("notify -> %ld\n", status);
    /* 202 表示已入队。响应体里每个目标一个投递 ID；改成在 payload 里加
     * "sync":true，就会阻塞到每个目标都有结果。 */

    /* ---------------------------------------------------------- 结果 */

    status = call(curl, "GET", "/api/v1/messages/282d67a1b3c4e5f6", auth, NULL);
    printf("status -> %ld\n", status);

    curl_easy_cleanup(curl);
    curl_global_cleanup();
    return 0;
}
